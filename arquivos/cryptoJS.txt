// crypto.js - Motor de Criptografia Total (Textos, Imagens, Áudios e Arquivos)
async function obterChaveSimetrica() {
    const senhaSecreta = "node_rex_chave_mestra_2026";
    const encoder = new TextEncoder();
    let keyData = new Uint8Array(32);
    const encodedSenha = encoder.encode(senhaSecreta);
    keyData.set(encodedSenha.subarray(0, 32));

    return await window.crypto.subtle.importKey(
        "raw",
        keyData,
        { name: "AES-GCM" },
        true,
        ["encrypt", "decrypt"]
    );
}

// Criptografa tanto texto puro quanto arquivos/imagens em formato string ou bytes
async function criptografarTexto(dadosPuros) {
    if (!dadosPuros) return "";
    const chave = await obterChaveSimetrica();
    const iv = window.crypto.getRandomValues(new Uint8Array(12));
    
    // Converte para bytes se for string (seja texto comum ou base64 de imagem/arquivo)
    const encoded = typeof dadosPuros === 'string' 
        ? new TextEncoder().encode(dadosPuros) 
        : dadosPuros;

    const ciphertext = await window.crypto.subtle.encrypt(
        { name: "AES-GCM", iv: iv },
        chave,
        encoded
    );

    const combined = new Uint8Array(iv.length + new Uint8Array(ciphertext).length);
    combined.set(iv);
    combined.set(new Uint8Array(ciphertext), iv.length);
    return btoa(String.fromCharCode(...combined));
}

// Descriptografa e reconstrói o texto ou o arquivo original perfeitamente
async function descriptografarTexto(textoCifradoBase64) {
    try {
        if (!textoCifradoBase64 || !textoCifradoBase64.startsWith("ENCRYPTED:")) {
            return textoCifradoBase64;
        }
        const limpo = textoCifradoBase64.replace("ENCRYPTED:", "");
        const chave = await obterChaveSimetrica();
        const combined = new Uint8Array(atob(limpo).split("").map(c => c.charCodeAt(0)));
        const iv = combined.slice(0, 12);
        const ciphertext = combined.slice(12);
        
        const decrypted = await window.crypto.subtle.decrypt(
            { name: "AES-GCM", iv: iv },
            chave,
            ciphertext
        );
        
        return new TextDecoder().decode(decrypted);
    } catch (e) {
        return "[Arquivo ou Mensagem criptografada / Chave incompatível]";
    }
}
