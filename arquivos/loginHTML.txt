    <script>
        const SEU_WHATSAPP = "5511999999999"; // Seu WhatsApp para receber ativações

        function mudarAba(aba) {
            document.querySelectorAll('.aba-btn').forEach(btn => btn.classList.remove('ativa'));
            document.querySelectorAll('.formulario').forEach(form => form.classList.remove('ativo'));

            if (aba === 'entrar') {
                document.querySelectorAll('.aba-btn')[0].classList.add('ativa');
                document.getElementById('formEntrar').classList.add('ativo');
            } else {
                document.querySelectorAll('.aba-btn')[1].classList.add('ativa');
                document.getElementById('formCadastrar').classList.add('ativo');
            }
        }

        function criarChavePorTelefone(nome, telefone) {
            const textoBase = (telefone + nome).replace(/\D/g, '').toLowerCase();
            let hash1 = 0;
            let hash2 = 0;

            for (let i = 0; i < textoBase.length; i++) {
                const char = textoBase.charCodeAt(i);
                hash1 = (hash1 << 5) - hash1 + char;
                hash1 |= 0;
                hash2 = (hash2 << 3) + hash2 ^ char;
                hash2 |= 0;
            }

            const p1 = Math.abs(hash1).toString(36).substring(0, 4).toUpperCase().padEnd(4, 'X');
            const p2 = Math.abs(hash2).toString(36).substring(0, 4).toUpperCase().padEnd(4, 'Y');

            return `DUO-${p1}-${p2}`;
        }

        function gerarNovaChave() {
            const nome = document.getElementById('cadNome').value.trim();
            const telefone = document.getElementById('cadTelefone').value.trim();
            const erroCad = document.getElementById('erroCad');

            erroCad.style.display = 'none';

            if (!nome || !telefone) {
                erroCad.innerText = "Por favor, preencha o Nome e o Telefone.";
                erroCad.style.display = 'block';
                return;
            }

            const chaveGerada = criarChavePorTelefone(nome, telefone);
            const senhaGerada = Math.floor(1000 + Math.random() * 9000).toString();

            document.getElementById('resChave').innerText = chaveGerada;
            document.getElementById('resSenha').innerText = senhaGerada;
            document.getElementById('resultadoChave').style.display = 'block';

            const texto = `Olá! Quero acesso ao node.REX++ (Chat Duo).%0A%0A*Nome:* ${nome}%0A*Telefone:* ${telefone}%0A*Chave:* ${chaveGerada}%0A*Senha:* ${senhaGerada}%0A%0APode liberar minha chave?`;
            window.open(`https://api.whatsapp.com/send?phone=${SEU_WHATSAPP}&text=${texto}`, '_blank');
        }

        async function autenticarUsuario() {
    const chave = document.getElementById('loginChave').value.trim();
    const senha = document.getElementById('loginSenha').value.trim();
    const erroLogin = document.getElementById('erroLogin');

    erroLogin.style.display = 'none';

    if (!chave || !senha) {
        erroLogin.innerText = "Preencha a chave e a senha.";
        erroLogin.style.display = 'block';
        return;
    }

    try {
        const resposta = await fetch('/api/login', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ chave, senha })
        });

        const resultado = await resposta.json();

        if (!resultado.sucesso) {
            erroLogin.innerText = resultado.erro;
            erroLogin.style.display = 'block';
            return;
        }

        // Salva a sessão localmente e entra no chat
        localStorage.setItem("sessao_node_rex", JSON.stringify({
            chave: chave,
            diasRestantes: resultado.diasRestantes
        }));

        window.location.href = "index.html";
    } catch (err) {
        erroLogin.innerText = "Erro ao conectar com o servidor.";
        erroLogin.style.display = 'block';
    }
}
